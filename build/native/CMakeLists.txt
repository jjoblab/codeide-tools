cmake_minimum_required(VERSION 3.14.2)
project(sdk-tools)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
enable_language(ASM)

# Port codeide (ADR 0012) — déclarations bionic absentes des en-têtes NDK
# curatés (__system_property_serial/area_serial : exportées par libc depuis
# API 19/21 et 23 mais jamais déclarées par le NDK, vérifié jusqu'à API 34) :
# incluses de force dans TOUTES les unités de compilation (les sources
# android-16 les utilisent à plusieurs endroits : libbase, liblog,
# libcutils/trace-dev.inc…).
add_compile_options(
    # C/CXX uniquement — le force-include d'un en-tête C dans les fichiers
    # ASSEMBLEUR (.S de boringssl) casse le préprocesseur asm.
    "$<$<COMPILE_LANGUAGE:C,CXX>:SHELL:-include ${CMAKE_SOURCE_DIR}/compat/include/bionic/system_properties_compat.h>"
)

# set global cflags and cxxflags
# -ftls-model=local-exec + -static (r3) : les binaires sont des exécutables
# ENTIÈREMENT STATIQUES (ET_EXEC, pas d'INTERP, pas de NEEDED) — le profil
# exact des binaires Lzhiyong 33-35 (« statically linked, for Android 30 »),
# qui passent sur tous les bancs et en production (ADR 0082). L'historique
# des échecs arm du banc Termux armv7 (bionic API 28) illustre pourquoi :
#   r1 (PIE dynamique)  : __tls_get_addr@LIBC_Q (GD TLS) — symbole API 29
#                         absent du bionic API 28 (run 37439785400) ;
#   r2 (PIE + IE TLS)   : R_ARM_TLS_TPOFF32 dynamique — reloc inconnue du
#                         chargeur de ce même banc (run 37449355987).
# Un exécutable statique ne traverse AUCUN chargeur : la question disparaît
# (le TLS local-exec est résolu au link, aucune reloc dynamique). La recette
# prévoyait ce mode (« static link (may need some patches) ») et le zip
# amont s'appelle android-sdk-tools-STATIC ; r1/r2 l'avaient oublié.
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fcolor-diagnostics -fPIC -Wno-attributes -ftls-model=local-exec")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fcolor-diagnostics -fPIC -Wno-attributes -ftls-model=local-exec")
# static link (may need some patches)
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -static")

# platform tools version
# see the patches/other/platform_tools_version.h
set(TOOLS_VERSION 35.0.2)
set(SRC ${PROJECT_SOURCE_DIR}/src)

# 64-bit off_t for lseek.
add_definitions(-D_FILE_OFFSET_BITS=64)

set(PROTOC_COMPILER)

if(DEFINED PROTOC_PATH)
    set(PROTOC_COMPILER ${PROTOC_PATH})
    
    if(NOT EXISTS ${PROTOC_COMPILER})
        unset(PROTOC_PATH CACHE)
        message(FATAL_ERROR "Invalid protoc: ${PROTOC_COMPILER}, please check if the path is correct")
    endif()
    
    # check protoc version
    execute_process(
        COMMAND ${PROTOC_COMPILER} --version
        OUTPUT_VARIABLE PTOTOBUF_VERSION
        WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    )
    
    if(NOT PTOTOBUF_VERSION MATCHES "^libprotoc*")
        unset(PROTOC_PATH CACHE)
        message(FATAL_ERROR "Invalid protoc: ${PROTOC_COMPILER}, this may not be an executable")
    endif()
    
    string(REGEX MATCH "([0-9]+)\\.([0-9]+)\\.([0-9]+)" VERSIONS ${PTOTOBUF_VERSION})
    # please note that if the protobuf version is too high, you may encounter the following error
    # error: This file was generated by a newer version of protoc 
    if(VERSIONS VERSION_GREATER "3.21.12")
        message(WARNING "${PTOTOBUF_VERSION}, the protobuf version is too high, may be incompatible!!")
    endif()
endif()

# building sdk-tools
# Port codeide (ADR 0012) — CODEIDE_SCOPE restreint la configuration :
#   all (défaut) : build-tools + platform-tools + others (zip complet) ;
#   build-tools  : exclut platform-tools/ + others/ (adb/fastboot exigent
#                  leur propre portage par tag — fichiers déplacés amont).
add_subdirectory(lib)
add_subdirectory(build-tools)
if(NOT DEFINED ENV{CODEIDE_SCOPE} OR "$ENV{CODEIDE_SCOPE}" STREQUAL "all")
    add_subdirectory(platform-tools)
    add_subdirectory(others)
endif()
